cmake_minimum_required(VERSION 3.16)
project(harurobo_teleop)

find_package(ament_cmake REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)

add_executable(teleop_node src/teleop_node.cpp)
target_compile_features(teleop_node PRIVATE cxx_std_17)
target_compile_options(teleop_node PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(teleop_node PRIVATE include)
ament_target_dependencies(teleop_node geometry_msgs rclcpp rcl_interfaces sensor_msgs std_msgs)
install(TARGETS teleop_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_manual_input test/test_manual_input.cpp)
  target_compile_features(test_manual_input PRIVATE cxx_std_17)
  target_include_directories(test_manual_input PRIVATE include)
endif()
ament_package()
