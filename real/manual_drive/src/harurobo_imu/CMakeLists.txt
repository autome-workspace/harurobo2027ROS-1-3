cmake_minimum_required(VERSION 3.16)
project(harurobo_imu)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
add_executable(imu_receiver src/imu_receiver.cpp)
target_compile_features(imu_receiver PRIVATE cxx_std_17)
target_compile_options(imu_receiver PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(imu_receiver PRIVATE include)
ament_target_dependencies(imu_receiver rclcpp rcl_interfaces sensor_msgs diagnostic_msgs)
install(TARGETS imu_receiver DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_imu_protocol test/test_imu_protocol.cpp)
  target_compile_features(test_imu_protocol PRIVATE cxx_std_17)
  target_include_directories(test_imu_protocol PRIVATE include)
endif()
ament_package()
