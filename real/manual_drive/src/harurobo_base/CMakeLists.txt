cmake_minimum_required(VERSION 3.16)
project(harurobo_base)

find_package(ament_cmake REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)

add_executable(can_monitor src/can_monitor.cpp src/socketcan_receiver.cpp)
target_compile_features(can_monitor PRIVATE cxx_std_17)
target_compile_options(can_monitor PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(can_monitor PRIVATE include)
ament_target_dependencies(can_monitor diagnostic_msgs rclcpp rcl_interfaces)
add_executable(heading_preview src/heading_preview.cpp)
target_compile_features(heading_preview PRIVATE cxx_std_17)
target_compile_options(heading_preview PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(heading_preview PRIVATE include)
ament_target_dependencies(heading_preview rclcpp rcl_interfaces sensor_msgs geometry_msgs diagnostic_msgs)
install(TARGETS heading_preview DESTINATION lib/${PROJECT_NAME})
install(TARGETS can_monitor DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_socketcan test/test_socketcan.cpp src/socketcan_receiver.cpp)
  target_compile_features(test_socketcan PRIVATE cxx_std_17)
  target_include_directories(test_socketcan PRIVATE include)
  ament_add_gtest(test_dm3520_speed_codec test/test_dm3520_speed_codec.cpp)
  target_compile_features(test_dm3520_speed_codec PRIVATE cxx_std_17)
  target_include_directories(test_dm3520_speed_codec PRIVATE include)
  ament_add_gtest(test_single_wheel_interlock test/test_single_wheel_interlock.cpp)
  target_compile_features(test_single_wheel_interlock PRIVATE cxx_std_17)
  target_include_directories(test_single_wheel_interlock PRIVATE include)
  ament_add_gtest(test_wheel_kinematics test/test_wheel_kinematics.cpp)
  target_compile_features(test_wheel_kinematics PRIVATE cxx_std_17)
  target_include_directories(test_wheel_kinematics PRIVATE include)
  ament_add_gtest(test_heading_hold test/test_heading_hold.cpp)
  target_compile_features(test_heading_hold PRIVATE cxx_std_17)
  target_include_directories(test_heading_hold PRIVATE include)
endif()
ament_package()
